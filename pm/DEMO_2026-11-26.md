```powershell
$env:STUDENTJOB_DB_PORT = '33481'
docker compose -p sj17demo1126 -f database/docker-compose.yml up -d --wait
Start-Process -FilePath (Get-Command node.exe).Source -ArgumentList 'demo-server.js' -WorkingDirectory (Join-Path (Get-Location) 'backend') -WindowStyle Hidden -PassThru
Start-Process -FilePath (Get-Command node.exe).Source -ArgumentList 'node_modules/vite/bin/vite.js','--host','127.0.0.1','--port','35173','--strictPort' -WorkingDirectory (Join-Path (Get-Location) 'frontend') -WindowStyle Hidden -PassThru
```
